# 课堂演示模式（demo mode）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax to track progress.

**Goal:** 让教师一键把系统按剧本自动走完 4 年 16 季，可在任意操作步骤暂停讲解、单步继续、跳到下一季，全程只跑在内存里、不写本机任何存档。

**Architecture:** 新增 `src/demo/` 四个模块——`script.ts` 声明每年的决策意图、`decide.ts` 用纯函数把意图 + 实时状态展开成动作描述符、`runner.ts` 把描述符映射到真实 store action 并按步进播放、`types.ts` 是三者共享契约。store 只加 `demoMode`/`demoSnapshot` 两个顶层字段和 `enterDemo`/`exitDemo` 两个动作，并在 `autoSaveGame` 首行短路。界面加一个右侧抽屉与顶部横幅，八个中心面板零改动。

**Tech Stack:** Next.js 16 / React 18 / TypeScript / Zustand 5 / Vitest 4（无新依赖）。

**Spec:** `docs/superpowers/specs/2026-10-12-demo-mode-design.md`（本计划逐节实现它；§9 的 7 条假设已与用户确认）

## Global Constraints

- 不新增任何 npm 依赖（规格 §7）。
- store 的校验失败一律 `return { validationError: '…' }`，不抛异常、不静默降级（沿用 `enterpriseStore.ts` 既有写法）。
- **演示期间不得写 `localStorage` 的 `enterpriseSaveFiles`**（规格 §4.8 三重闸）——这是本功能唯一的不可协商项，roadmap §3 P0 记过一次真实的存档覆盖事故。
- 讲解卡与横幅的文案不得新起一份：规则文字一律引用 `RULE_SECTIONS`，步骤名一律引用 `CONTROL_STEPS`（规格 §4.3；roadmap §5.3 记过"以显示名为键、改名即静默失效"）。
- 金额单位为 M，整数；不引入 P3/P4 相关入口。
- `directions/` 永不改动。
- 提交信息：中文 Conventional Commits，与仓库现有风格一致（如 `feat(demo): …`）。
- **测试必须在 worktree 目录内跑**：主检出会把 `.worktrees/**` 里的同一批测试再扫一遍，报出双倍的假数（roadmap §5.9）。每个任务收尾门槛是 `npx vitest run` 全绿；涉及 UI 的任务再加 `npm run build` 通过。
- 自动播放最快档不得低于 300ms/步（规格 §4.6：低于它就会撞进账实核对的同毫秒盲区，导出的演示局会印出一条不存在的"账实不符"）。

```ts
/** 按讲解点取回规则原文；取不到返回 null，由调用方退化显示步骤名（规格 §5） */
export const ruleTextFor = (ref: RuleRef): string | null => {

};
```

```ts
// 意图 + 实时状态 → 动作序列（纯函数，规格 §4.5）。这里不做任何状态写入，也不 import store：
// 一旦 import store 就没法用 createFreshState() 造状态单测它了。
import type { EnterpriseState } from '../types/enterprise';
import { PRODUCT_BOM, type MaterialId } from '../utils/rules';
import type { DemoStep, PlayableProduct, YearIntent } from './types';

export const linesOf = (s: EnterpriseState) =>
  s.production.factories.flatMap((f) => f.productionLines);

/**
 * 本年还能承诺多少件：现有成品 + 各线（含在手在制品）到年底的完工量 − 已选未交。
 * 全自动线 productionPeriod=1 → 每季 1 件；余量按整除算，宁少不多（选多了交不出会撞库存门）。
 */
export const commitableUnits = (s: EnterpriseState, product: PlayableProduct): number => {
  const quartersLeft = 4 - s.operation.currentQuarter + 1;
  const makeable = linesOf(s)
    .filter((l) => l.product === product && l.status !== 'selling')
    .reduce((sum, l) => sum + l.inProgressProducts + Math.floor(quartersLeft / l.productionPeriod), 0);
  const stock = s.logistics.finishedProducts.find((p) => p.type === product)?.quantity ?? 0;
  const promised = s.marketing.selectedOrders
    .filter((o) => !o.isDelivered && o.productType === product)
    .reduce((sum, o) => sum + o.quantity, 0);
  return stock + makeable - promised;
};

/** 单价高优先，同价小单优先、账期短优先；累计量受可承诺量与 orderCap 双重约束 */
export const pickOrders = (s: EnterpriseState, intent: YearIntent): string[] => {
  const sorted = [...s.marketing.availableOrders].sort((a, b) =>
    b.unitPrice - a.unitPrice || a.quantity - b.quantity || a.paymentPeriod - b.paymentPeriod);
  const room: Record<PlayableProduct, number> = {
    P1: commitableUnits(s, 'P1'),
    P2: commitableUnits(s, 'P2'),
  };
  const chosen: string[] = [];
  for (const o of sorted) {
    if (chosen.length >= intent.orderCap) break;
    const product = o.productType as PlayableProduct;
    if (product !== 'P1' && product !== 'P2') continue;   // 本期无 P3/P4 运营入口
    if (room[product] >= o.quantity) {
      room[product] -= o.quantity;
      chosen.push(o.id);
    }
  }
  return chosen;
};

export const deliverySteps = (s: EnterpriseState): DemoStep[] =>
  s.marketing.selectedOrders
    .filter((o) => !o.isDelivered)
    .filter((o) => (s.logistics.finishedProducts.find((p) => p.type === o.productType)?.quantity ?? 0) >= o.quantity)
    .map((o) => ({ stepId: 'q-14', op: { kind: 'deliverOrder', orderId: o.id } as DemoStep['op'] }));
```

```ts
/** 每季投料需求（按各线当前产品 BOM 汇总） */
export const quarterlyMaterialDemand = (s: EnterpriseState): Record<MaterialId, number> => {
  const demand: Record<MaterialId, number> = { R1: 0, R2: 0, R3: 0, R4: 0 };
  for (const line of linesOf(s)) {
    if (!line.product || line.status === 'selling') continue;
    for (const [type, qty] of Object.entries(PRODUCT_BOM[line.product as MaterialProductKey])) {
      demand[type as MaterialId] += qty ?? 0;
    }
  }
  return demand;
};

// PRODUCT_BOM 的键类型（P1..P4）；本函数只按已开放的产品取料
type MaterialProductKey = keyof typeof PRODUCT_BOM;

/** 下原料订单：到货入库时才付款（store:1516），所以这里只看数量缺口，不看现金 */
export const materialSteps = (s: EnterpriseState): DemoStep[] => {
  const demand = quarterlyMaterialDemand(s);
  const out: DemoStep[] = [];
  for (const type of ['R1', 'R2', 'R3', 'R4'] as MaterialId[]) {
    const need = demand[type];
    if (need <= 0) continue;
    const material = s.logistics.rawMaterials.find((m) => m.type === type);
    const leadTime = material?.leadTime ?? 1;
    const onHand = material?.quantity ?? 0;
    const pending = s.logistics.rawMaterialOrders
      .filter((o) => o.materialType === type)
      .reduce((sum, o) => sum + o.quantity, 0);
    const shortfall = need * (leadTime + 1) - onHand - pending;
    if (shortfall > 0) {
      out.push({ stepId: 'q-6', op: { kind: 'placeRawMaterialOrder', materialType: type, quantity: shortfall } });
    }
  }
  return out;
};

/**
 * q-8 上新线（本年最多落一条）+ q-10 开工。
 * 在建判定用 status==='installing' 或 builtInYear===当年：安装完成的下一季度才可生产（规则弹窗原文），
 * 所以本年已投的线不该再投一条，否则 orderCap 之外还会多养一条线。
 */
export const productionSteps = (s: EnterpriseState, intent: YearIntent): DemoStep[] => {
  const out: DemoStep[] = [];
  for (const plan of intent.newLines) {
    const factory = s.production.factories.find((f) => f.id === plan.factoryId);
    if (!factory || factory.productionLines.length >= factory.capacity) continue;
    const wanted = intent.newLines.filter((p) => p.factoryId === plan.factoryId).length;
    const started = factory.productionLines.filter(
      (l) => l.status === 'installing' || l.builtInYear === s.operation.currentYear,
    ).length;
    if (started < wanted) {
      out.push({
        stepId: 'q-8',
        op: { kind: 'addProductionLine', factoryId: plan.factoryId, lineType: plan.lineType, product: plan.product },
      });
      break;   // 每季最多落一条：分期投资要占现金，一季落两条会撞加工费/首期门
    }
  }

  for (const line of linesOf(s)) {
    if (!line.product || line.inProgressProducts > 0) continue;
    if (line.status === 'installing' || line.status === 'converting' || line.status === 'selling') continue;
    const product = line.product as PlayableProduct;
    if (product !== 'P1' && product !== 'P2') continue;
    const stocked = Object.entries(PRODUCT_BOM[product]).every(
      ([type, qty]) =>
        (s.logistics.rawMaterials.find((m) => m.type === type)?.quantity ?? 0) >= (qty ?? 0),
    );
    if (stocked) out.push({ stepId: 'q-10', op: { kind: 'startProduction', lineId: line.id } });
  }
  return out;
};
```

`materialSteps` 的料齐判定里 `stocked` 只看单条线的 BOM 用量、不看"两条线争同一个料"，这是刻意的保守：真争抢会让第二条线撞 store:1249 的"原材料不足"，runner 停下来说得清——比在决策层提前复算库存分配多一套会漂的规则要好。

```ts
/**
 * 跨入"下一季"那一刻引擎会自动扣掉的钱（规格 §4.5）。刻意只列**引擎会自己扣**的项：
 * 手动动作（加工费、广告、市场投资）各有自己的现金门，撞了由 runner 报出来，不在这里复算——
 * 复算 `nextQuarter` 那 790 行只会造出第二套会漂的规则。缺一项的后果是演示停下来说得出缺口，
 * 多算一项的后果只是演示保守一点。
 */
export const estAutoOutlay = (s: EnterpriseState): number => {
  const nextAbs = absQuarter(s.operation.currentYear, s.operation.currentQuarter) + 1;
  const { year, quarter } = fromAbsQuarter(nextAbs);
  const crossesYearEnd = quarter === 1;   // 跨入第1季 = 刚结算完上一年的年末

  const shortDue = settleDueShortLoans(s.finance.loans, nextAbs).due;
  const long = crossesYearEnd ? settleLongLoansAtYearEnd(s.finance.loans) : { interest: 0, principalRepaid: 0 };
  const rent = crossesYearEnd ? annualRent(s.production.factories) : 0;
  // 维护费免当年建成（store:2611 用 builtInYear < closingYear），closingYear 即刚结束那年
  const maintenance = crossesYearEnd
    ? linesOf(s).filter((l) => l.builtInYear < year - 1).reduce((sum, l) => sum + l.maintenanceCost, 0)
    : 0;
  const admin = quarter === 4 ? 1 : 0;
  const rd = s.production.productRD.P2.status === 'active' ? 1 : 0;
  const payable = crossesYearEnd ? s.finance.accountsPayable : 0;
  const arriving = s.logistics.rawMaterialOrders
    .filter((o) => o.arrivalPeriod === nextAbs)
    .reduce((sum, o) => sum + o.quantity * o.price, 0);

  return shortDue + long.interest + long.principalRepaid + rent + maintenance + admin + rd + payable + arriving;
};

/** 季末现金垫：预算 + 剧本基线垫；贴现取"够到它的最小 7 的倍数"，救不回就不动（规格 §9-5 后备顺序第 1 档） */
export const rescueSteps = (s: EnterpriseState, intent: YearIntent): DemoStep[] => {
  const floor = estAutoOutlay(s) + intent.cashPad;
  if (s.finance.cash >= floor) return [];
  const totalAR = s.finance.accountsReceivable.reduce((sum, v) => sum + v, 0);
  const max = Math.floor(totalAR / DISCOUNT_UNIT) * DISCOUNT_UNIT;
  for (let amount = DISCOUNT_UNIT; amount <= max; amount += DISCOUNT_UNIT) {
    const gain = amount - amount / DISCOUNT_UNIT;   // 与 rules.discountSplit 同一算法，7M 付 1M 贴息
    if (s.finance.cash + gain >= floor) {
      return [{ stepId: 'q-11', op: { kind: 'discountReceivable', amount } }];
    }
  }
  return [];
};
```

```ts
  const steps: DemoStep[] = [];

  if (intent.shortLoanQuarters.includes(quarter)) steps.push({ stepId: 'q-3', op: { kind: 'applyShortTermLoan' } });
  steps.push(...materialSteps(s));
  steps.push(...productionSteps(s, intent));
  steps.push(...deliverySteps(s));
  if (intent.rdP2 && s.production.productRD.P2.status === 'idle') {
    steps.push({ stepId: 'q-15', op: { kind: 'investProductR_D' } });
  }
  if (quarter === 4) steps.push(...yearEndSteps(s, intent));
  steps.push(...rescueSteps(s, intent));
  steps.push({ stepId: 'q-20', op: { kind: 'nextQuarter' } });

  return steps;
};
```

```ts
// 讲解点"只自动停一次"靠它，模块级即可：start() 会清空
let seenPausePoints = new Set<string>();

const expand = (): RunningStep[] => {
  const s = useEnterpriseStore.getState().state;
  if (s.operation.isGameOver) return [];
  const intent = YEAR_INTENTS[s.operation.currentYear as 1 | 2 | 3 | 4];
  return decideQuarter(s, intent).map((d) => {
    const point = pausePointFor(d.stepId);
    return {
      ...d,
      name: stepName(d.stepId),
      label: describeOp(d.op),
      rule: point?.rule,
      isPausePoint: !!point && !seenPausePoints.has(d.stepId),
    };
  });
};

/**
 * 执行一步，返回是否被拒。判据用 **state 引用**而不是 validationError：
 * 成功路径都返回新的 state 对象，被拒路径只塞一句 validationError（store 各门的既有写法），
 * 所以"引用没变"才是结构性的失败信号；validationError 只负责提供那句人话原因。
 * store 里还有 `return state` 式的静默守卫（如 :1497 原料类型找不到、:2016 订单已交），
 * 它们不改引用也不报错——那种情况自己造一句原因，别让它静默滑过去。
 */
const advanceOne = (): { ok: boolean } => {
  const runner = useDemoRunner.getState();
  const [head, ...rest] = runner.queue;
  // 队列空掉只可能是编排漏了推进步：宁可停出来说话，也不要在这里偷偷 expand() 重跑一遍本季
  // （重跑会二次投广告、二次下单，那是数据损坏而不是停住）
  if (!head) {
    const op = useEnterpriseStore.getState().state.operation;
    useDemoRunner.setState({
      phase: 'blocked',
      blocked: { year: op.currentYear, quarter: op.currentQuarter, label: '本季动作序列', reason: '队列已空但没走到推进步，编排层有 bug' },
    });
    return { ok: false };
  }

  const store = useEnterpriseStore.getState();
  const { currentYear, currentQuarter } = store.state.operation;
  store.setValidationError(null);
  const before = useEnterpriseStore.getState().state;
  dispatch(head.op);
  const after = useEnterpriseStore.getState().state;

  if (after === before) {
    const reason = useEnterpriseStore.getState().validationError
      ?? '此步未改动任何状态（store 的静默守卫，例如目标已不存在或已交付）';
    useDemoRunner.setState({
      phase: 'blocked',
      blocked: { year: currentYear, quarter: currentQuarter, label: `${head.name}｜${head.label}`, reason },
      queue: [head, ...rest],   // 保留这一步，"重试此步"才有的可重
    });
    return { ok: false };
  }

  if (head.isPausePoint) seenPausePoints = new Set([...seenPausePoints, head.stepId]);

  // nextQuarter 之后必须换季重算；一季跑完（rest 空）也重算，防止剧本在某季多产步
  const finishedGame = head.op.kind === 'nextQuarter'
    && useEnterpriseStore.getState().state.operation.isGameOver;
  const nextQueue = head.op.kind === 'nextQuarter' || rest.length === 0 ? expand() : rest;

  if (finishedGame) {
    useDemoRunner.setState({ queue: [], phase: 'finished', blocked: null });
    return { ok: true };
  }
  useDemoRunner.setState({
    queue: nextQueue,
    blocked: null,
    phase: head.isPausePoint ? 'paused' : useDemoRunner.getState().phase,
  });
  return { ok: true };
};
```

```ts
    const tick = () => {
      const r = advanceOne();
      if (!r.ok || useDemoRunner.getState().phase === 'finished') { clearTimer(); return; }
      if (useDemoRunner.getState().phase === 'paused') { clearTimer(); return; }   // 讲解点到了就停
      timer = setTimeout(tick, Math.max(MIN_SPEED_MS, get().speedMs));
    };
    timer = setTimeout(tick, Math.max(MIN_SPEED_MS, get().speedMs));
  },

  pause: () => { clearTimer(); set({ phase: 'paused' }); },

  // 跳到下一季：把当季剩余步跑完，年/季一变就停；撞门也停在那儿让老师看原因。
  // guard 计数是必要的兜底：nextQuarter 那一步永远排在本季末尾（decideQuarter 保证），所以季界变化必然发生，
  // 但剧本万一在某季产出了空队列，没有这道闸这个 while 就转不完。
  skipQuarter: () => {
    clearTimer();
    const startAbs = absQuarter(
      useEnterpriseStore.getState().state.operation.currentYear,
      useEnterpriseStore.getState().state.operation.currentQuarter,
    );
    for (let guard = 0; guard < 100; guard++) {
      const r = useDemoRunner.getState();
      if (r.phase === 'blocked' || r.phase === 'finished' || r.queue.length === 0) return;
      advanceOne();
      const now = useEnterpriseStore.getState().state.operation;
      if (absQuarter(now.currentYear, now.currentQuarter) !== startAbs) return;
    }
  },

  restart: () => { clearTimer(); get().start(); },

  exit: () => { clearTimer(); useEnterpriseStore.getState().exitDemo(); set({ phase: 'idle', queue: [], blocked: null }); },

  setSpeed: (ms) => set({ speedMs: Math.max(MIN_SPEED_MS, ms) }),
}));
```

`play()` 的停止条件有三个：撞门、跑完、走到讲解点（`advanceOne` 把 `phase` 置成 `paused`）。讲解点因此是"自动播也会停"的地方，不是只有手动模式才停。
